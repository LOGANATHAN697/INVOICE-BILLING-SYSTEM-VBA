Private Sub AddCompanyLogoInPDF(ByVal ws As Worksheet)

    Dim logoPic As Shape

    If Dir(COMPANY_LOGO_PATH) = "" Then
        MsgBox "Logo file not found:" & vbCrLf & COMPANY_LOGO_PATH, _
               vbExclamation, "Logo Missing"
        Exit Sub
    End If

    On Error Resume Next
    ws.Shapes("Company_Logo").Delete
    On Error GoTo Logo_Error

    Set logoPic = ws.Shapes.AddPicture( _
        fileName:=COMPANY_LOGO_PATH, _
        LinkToFile:=msoFalse, _
        SaveWithDocument:=msoTrue, _
        Left:=ws.Range("d4").Left + 4, _
        Top:=ws.Range("d4").Top + 1, _
        Width:=400, _
        Height:=250 _
    )

    With logoPic
        .Name = "Company_Logo"
        .Placement = xlMoveAndSize
        .LockAspectRatio = msoFalse
        .ZOrder msoBringToFront
    End With

    Exit Sub

Logo_Error:
    MsgBox "Logo insert failed." & vbCrLf & _
           "Reason: " & Err.Description, _
           vbExclamation, "Logo Error"

End Sub

Private Sub KeepOnlyDigits(ByVal txt As MSForms.TextBox, Optional ByVal maxLen As Long = 0)

    Dim oldText As String
    Dim newText As String
    Dim i As Long
    Dim ch As String

    If isCleaningText Then Exit Sub
    If txt Is Nothing Then Exit Sub

    oldText = txt.Text

    For i = 1 To Len(oldText)
        ch = Mid$(oldText, i, 1)
        If ch >= "0" And ch <= "9" Then
            newText = newText & ch
        End If
    Next i

    If maxLen > 0 Then
        newText = Left$(newText, maxLen)
    End If

    If oldText <> newText Then
        isCleaningText = True
        txt.Text = newText
        txt.SelStart = Len(txt.Text)
        isCleaningText = False
    End If

End Sub

Private Function ValidatePhoneNumber() As Boolean

    Dim phoneNo As String

    phoneNo = Trim(Me.Controls("txtPhone").Text)
    phoneNo = Replace(phoneNo, " ", "")
    phoneNo = Replace(phoneNo, "-", "")

    If phoneNo = "" Then
        MsgBox "Must be a 10-digit number", vbExclamation, "Invalid Phone Number"
        Me.Controls("txtPhone").SetFocus
        ValidatePhoneNumber = False
        Exit Function
    End If

    If Len(phoneNo) <> 10 Then
        MsgBox "Must be a 10-digit number", vbExclamation, "Invalid Phone Number"
        Me.Controls("txtPhone").SetFocus
        ValidatePhoneNumber = False
        Exit Function
    End If

    If Not phoneNo Like "##########" Then
        MsgBox "Must be a 10-digit number", vbExclamation, "Invalid Phone Number"
        Me.Controls("txtPhone").SetFocus
        ValidatePhoneNumber = False
        Exit Function
    End If

    Me.Controls("txtPhone").Text = phoneNo
    ValidatePhoneNumber = True

End Function

Private Function IsProductInList(ByVal productName As String) As Boolean

    Dim i As Long
    Dim lst As MSForms.ListBox

    On Error Resume Next
    Set lst = Me.Controls("lstItems")
    On Error GoTo 0

    If lst Is Nothing Then Exit Function

    With lst
        For i = 0 To .ListCount - 1
            If StrComp(Trim$(.List(i, 1)), Trim$(productName), vbTextCompare) = 0 Then
                IsProductInList = True
                Exit Function
            End If
        Next i
    End With

End Function

Private Function GetNextInvoiceNo() As String

    Dim ws As Worksheet
    Dim lastRow As Long
    Dim nextNo As Long

    Set ws = ThisWorkbook.Worksheets("Invoice_Data")

    lastRow = ws.Cells(ws.Rows.Count, "A").End(xlUp).Row

    If lastRow < 2 Then
        nextNo = 1
    Else
        nextNo = lastRow
    End If

    GetNextInvoiceNo = "INV-" & Format(nextNo, "0000")

End Function

Private Sub CalculateSingleLineAmount()

    Dim qty As Double
    Dim rate As Double
    Dim gstPercent As Double
    Dim amountBeforeGST As Double
    Dim gstAmount As Double
    Dim totalAmount As Double
    Dim paidAmount As Double
    Dim balanceAmount As Double

    qty = Val(Me.Controls("txtQty").Text)
    rate = Val(Me.Controls("txtRate").Text)
    gstPercent = Val(Me.Controls("txtGST").Text)

    amountBeforeGST = qty * rate
    gstAmount = amountBeforeGST * gstPercent / 100
    totalAmount = amountBeforeGST + gstAmount

Private Sub AddProductItem()

    Dim productName As String
    Dim qty As Double
    Dim rate As Double
    Dim gstPercent As Double
    Dim amountBeforeGST As Double
    Dim gstAmount As Double
    Dim totalWithGST As Double
    Dim rowNo As Long

    productName = Trim(Me.Controls("txtProduct").Value)
    qty = Val(Me.Controls("txtQty").Text)
    rate = Val(Me.Controls("txtRate").Text)
    gstPercent = Val(Me.Controls("txtGST").Value)

    If productName = "" Then
        MsgBox "Please select product", vbExclamation, "Product"
        Me.Controls("txtProduct").SetFocus
        Exit Sub
    End If

    If qty <= 0 Then
        MsgBox "Please enter quantity", vbExclamation, "Qty"
        Me.Controls("txtQty").SetFocus
        Exit Sub
    End If

    If rate <= 0 Then
        MsgBox "Rate missing. Please select product again.", vbExclamation, "Rate"
        Me.Controls("txtProduct").SetFocus
        Exit Sub
    End If

    If IsProductInList(productName) Then
        MsgBox "This product already added.", vbExclamation, "Duplicate Product"
        Exit Sub
    End If

    amountBeforeGST = qty * rate
    gstAmount = amountBeforeGST * gstPercent / 100
    totalWithGST = amountBeforeGST + gstAmount

    With Me.Controls("lstItems")
        .AddItem CStr(.ListCount + 1)
        rowNo = .ListCount - 1

        .List(rowNo, 2) = Format(qty, "0")

        .List(rowNo, 5) = Format(gstPercent, "0.00")
        .List(rowNo, 6) = Format(gstAmount, "0.00")
        .List(rowNo, 7) = Format(totalWithGST, "0.00")

        .ListIndex = rowNo
        .ZOrder 0
    End With

    Me.Controls("txtQty").Text = ""
    Me.Controls("txtRate").Text = ""

    Call RecalculateMultipleItemsTotal
    Call LoadProductDropdown

End Sub

Private Function GetProductNameFromSummary(ByVal summaryText As String) As String

    Dim pos As Long

    summaryText = Trim(summaryText)
    pos = InStrRev(summaryText, " x ")

    If pos > 0 Then
        GetProductNameFromSummary = Trim(Left(summaryText, pos - 1))
    Else
        GetProductNameFromSummary = summaryText
    End If

End Function

Private Function GetQtyFromSummary(ByVal summaryText As String) As Double

    Dim pos As Long
    Dim qtyText As String

    summaryText = Trim(summaryText)
    pos = InStrRev(summaryText, " x ")

    If pos > 0 Then
        qtyText = Trim(Mid(summaryText, pos + 3))
        GetQtyFromSummary = Val(qtyText)
    Else
        GetQtyFromSummary = 1
    End If

    If GetQtyFromSummary <= 0 Then GetQtyFromSummary = 1

End Function

Private Sub LoadInvoiceItems(ByVal invoiceNo As String)

    Dim itemWs As Worksheet
    Dim dataWs As Worksheet
    Dim lastRow As Long
    Dim r As Long
    Dim rowNo As Long
    Dim invoiceRow As Long
    Dim loadedCount As Long

    Dim productSummary As String
    Dim productName As String
    Dim qty As Double
    Dim totalWithGST As Double
    Dim paidAmount As Double
    Dim balanceAmount As Double
    Dim gstPercent As Double
    Dim amountBeforeGST As Double
    Dim gstAmount As Double
    Dim rate As Double

    On Error Resume Next
    Me.Controls("lstItems").Clear
    On Error GoTo 0

    Set itemWs = ThisWorkbook.Worksheets("Invoice_Items")
    lastRow = itemWs.Cells(itemWs.Rows.Count, "A").End(xlUp).Row

    'Load from Invoice_Items sheet
    For r = 2 To lastRow

        If Trim(CStr(itemWs.Cells(r, 1).Value)) = Trim(CStr(invoiceNo)) Then

            With Me.Controls("lstItems")
                .AddItem ""
                rowNo = .ListCount - 1

                .List(rowNo, 0) = rowNo + 1
                .List(rowNo, 1) = itemWs.Cells(r, 3).Value
                .List(rowNo, 2) = Format(Val(itemWs.Cells(r, 4).Value), "0")
                .List(rowNo, 3) = Format(Val(itemWs.Cells(r, 5).Value), "0.00")
                .List(rowNo, 4) = Format(Val(itemWs.Cells(r, 6).Value), "0.00")
                .List(rowNo, 5) = Format(Val(itemWs.Cells(r, 7).Value), "0.00")
                .List(rowNo, 6) = Format(Val(itemWs.Cells(r, 8).Value), "0.00")
                .List(rowNo, 7) = Format(Val(itemWs.Cells(r, 9).Value), "0.00")
            End With

            loadedCount = loadedCount + 1

        End If

    Next r

    If loadedCount > 0 Then
        Call ReNumberItems
        Exit Sub
    End If

    'Fallback: older invoice saved only in Invoice_Data
    Set dataWs = ThisWorkbook.Worksheets("Invoice_Data")
    invoiceRow = FindInvoiceRow(invoiceNo)

    If invoiceRow = 0 Then Exit Sub

    productSummary = Trim(CStr(dataWs.Cells(invoiceRow, 9).Value))

    If productSummary = "" Then Exit Sub

    productName = GetProductNameFromSummary(productSummary)
    qty = GetQtyFromSummary(productSummary)

    totalWithGST = Val(dataWs.Cells(invoiceRow, 11).Value)
    paidAmount = Val(dataWs.Cells(invoiceRow, 12).Value)
    balanceAmount = Val(dataWs.Cells(invoiceRow, 13).Value)

    If totalWithGST <= 0 Then totalWithGST = paidAmount + balanceAmount

    gstPercent = Val(Me.Controls("txtGST").Value)
    If gstPercent <= 0 Then gstPercent = 18

    amountBeforeGST = totalWithGST / (1 + (gstPercent / 100))
    gstAmount = totalWithGST - amountBeforeGST

    If qty <= 0 Then qty = 1
    rate = amountBeforeGST / qty

    With Me.Controls("lstItems")
        .AddItem ""
        rowNo = .ListCount - 1

        .List(rowNo, 0) = rowNo + 1
        .List(rowNo, 1) = productName
        .List(rowNo, 2) = Format(qty, "0")
        .List(rowNo, 3) = Format(rate, "0.00")
        .List(rowNo, 4) = Format(amountBeforeGST, "0.00")
        .List(rowNo, 5) = Format(gstPercent, "0.00")
        .List(rowNo, 6) = Format(gstAmount, "0.00")
        .List(rowNo, 7) = Format(totalWithGST, "0.00")
    End With

    Call ReNumberItems

End Sub

Private Sub GenerateInvoicePDF()

    Dim ws As Worksheet
    Dim pdfFolder As String
    Dim pdfPath As String
    Dim invoiceNo As String
    Dim customerName As String

    On Error GoTo PDF_Error

    If Trim(Me.Controls("txtInvoiceNo").Text) = "" Then
        MsgBox "Invoice No is missing", vbCritical, "PDF Error"
        Exit Sub
    End If

    If Trim(Me.Controls("txtCustomerName").Text) = "" Then
        MsgBox "Enter Customer Name before generating PDF", vbCritical, "PDF Error"
        Me.Controls("txtCustomerName").SetFocus
        Exit Sub
    End If

    If Not EnsureItemsAvailable() Then Exit Sub
    

    Call RecalculateMultipleItemsTotal

    invoiceNo = CleanFileName(Me.Controls("txtInvoiceNo").Text)
    customerName = CleanFileName(Me.Controls("txtCustomerName").Text)

    If ThisWorkbook.Path = "" Then
        pdfFolder = Environ$("USERPROFILE") & "\Desktop"
    Else
        pdfFolder = ThisWorkbook.Path
    End If

    pdfPath = pdfFolder & "\" & invoiceNo & "_" & customerName & ".pdf"

    If Dir(pdfPath) <> "" Then
        On Error Resume Next
        Kill pdfPath

        If Err.Number <> 0 Then
            MsgBox "Please close the already opened PDF file and try again:" & vbCrLf & pdfPath, _
                   vbCritical, "PDF Already Open"
            Err.Clear
            Exit Sub
        End If

        On Error GoTo PDF_Error
    End If

        'Set ws = GetInvoicePrintSheet()
        Set ws = GetInvoicePrintSheet()

        Call PrepareInvoicePrintSheet(ws)
        Call AddCompanyLogoInPDF(ws)
        
        ''Call AddCompanyLogoFromGoogleDrive(ws)
        ws.Activate

    Application.DisplayAlerts = False

    ws.ExportAsFixedFormat _
        Type:=xlTypePDF, _
        fileName:=pdfPath, _
        Quality:=xlQualityStandard, _
        IncludeDocProperties:=True, _
        IgnorePrintAreas:=False, _
        OpenAfterPublish:=True

    Application.DisplayAlerts = True

    MsgBox "PDF invoice created successfully:" & vbCrLf & pdfPath, vbInformation, "PDF Created"

    Exit Sub

PDF_Error:

    Application.DisplayAlerts = True

    MsgBox "PDF export failed." & vbCrLf & _
           "Reason: " & Err.Description & vbCrLf & vbCrLf & _
           "Please close opened PDF file and try again.", _
           vbCritical, "PDF Error"

End Sub



Private Sub AddInvoicePDFExtraDetails(ByVal ws As Worksheet)

    Dim totalAmount As Double
    Dim amountWords As String
    Dim startRow As Long
    Dim tickMark As String

    tickMark = ChrW(&H2713)

    totalAmount = GetInvoiceTotalAmount(ws)
    amountWords = AmountToWordsIndian(totalAmount)

    startRow = ws.Cells(ws.Rows.Count, "A").End(xlUp).Row + 2

    If startRow < 38 Then startRow = 38

    With ws
        .Range("A" & startRow & ":H" & startRow + 5).Clear

        .Range("A" & startRow & ":C" & startRow).Merge
        .Range("A" & startRow).Value = "Original for Recipient"
        .Range("A" & startRow).Font.Bold = True
        .Range("A" & startRow).Font.Size = 12
        .Range("A" & startRow).HorizontalAlignment = xlLeft

        .Range("A" & startRow + 1 & ":H" & startRow + 2).Merge

Private Function GetInvoiceTotalAmount(ByVal ws As Worksheet) As Double

    Dim amountText As String
    Dim cell As Range
    Dim i As Long
    Dim labels As Variant

    amountText = ControlTextSafe("txtGrandTotal")
    If CleanAmount(amountText) > 0 Then
        GetInvoiceTotalAmount = CleanAmount(amountText)
        Exit Function
    End If

    amountText = ControlTextSafe("txtTotalAmount")
    If CleanAmount(amountText) > 0 Then
        GetInvoiceTotalAmount = CleanAmount(amountText)
        Exit Function
    End If

    amountText = ControlTextSafe("txtNetAmount")
    If CleanAmount(amountText) > 0 Then
        GetInvoiceTotalAmount = CleanAmount(amountText)
        Exit Function
    End If

    amountText = ControlTextSafe("txtFinalTotal")
    If CleanAmount(amountText) > 0 Then
        GetInvoiceTotalAmount = CleanAmount(amountText)
        Exit Function
    End If

    labels = Array("grand total", "total amount", "net amount", "invoice total")

    For Each cell In ws.UsedRange.Cells
        If Not IsError(cell.Value) Then
            For i = LBound(labels) To UBound(labels)
                If InStr(1, LCase(CStr(cell.Value)), labels(i), vbTextCompare) > 0 Then
                    GetInvoiceTotalAmount = CleanAmount(CStr(cell.Offset(0, 1).Value))
                    If GetInvoiceTotalAmount = 0 Then
                        GetInvoiceTotalAmount = CleanAmount(CStr(cell.Offset(0, 2).Value))
                    End If
                    If GetInvoiceTotalAmount > 0 Then Exit Function
                End If
            Next i
        End If
    Next cell

    GetInvoiceTotalAmount = 0

End Function


Private Function ControlTextSafe(ByVal controlName As String) As String

    On Error Resume Next

    ControlTextSafe = Me.Controls(controlName).Text

    If Err.Number <> 0 Then
        Err.Clear
        ControlTextSafe = Me.Controls(controlName).Value
    End If

    If Err.Number <> 0 Then
        Err.Clear
        ControlTextSafe = Me.Controls(controlName).Caption
    End If

    If Err.Number <> 0 Then
        Err.Clear
        ControlTextSafe = ""
    End If

    On Error GoTo 0

End Function


Private Function CleanAmount(ByVal amountText As String) As Double

    Dim s As String

    s = CStr(amountText)
    s = Replace(s, "?", "")
    s = Replace(s, "Rs.", "")
    s = Replace(s, "Rs", "")
    s = Replace(s, "INR", "")
    s = Replace(s, ",", "")
    s = Replace(s, " ", "")

    If IsNumeric(s) Then
        CleanAmount = CDbl(s)
    Else
        CleanAmount = Val(s)
    End If

End Function


Private Function AmountToWordsIndian(ByVal amount As Double) As String

    Dim rupees As Long
    Dim paise As Long
    Dim result As String

    rupees = Fix(amount)
    paise = Round((amount - rupees) * 100, 0)

    If paise = 100 Then
        rupees = rupees + 1
        paise = 0
    End If

    result = "Rupees " & IndianNumberToWords(rupees)

    If paise > 0 Then
        result = result & " and " & IndianNumberToWords(paise) & " Paise"
    End If

    AmountToWordsIndian = result & " Only"

End Function


Private Function IndianNumberToWords(ByVal numberValue As Long) As String

    Dim crore As Long
    Dim lakh As Long
    Dim thousand As Long
    Dim hundred As Long
    Dim result As String

    If numberValue = 0 Then
        IndianNumberToWords = "Zero"
        Exit Function
    End If

    crore = numberValue \ 10000000
    numberValue = numberValue Mod 10000000

    lakh = numberValue \ 100000
    numberValue = numberValue Mod 100000

    thousand = numberValue \ 1000
    numberValue = numberValue Mod 1000

    hundred = numberValue

    If crore > 0 Then
        result = result & NumberBelowThousand(crore) & " Crore "
    End If

    If lakh > 0 Then
        result = result & NumberBelowThousand(lakh) & " Lakh "
    End If

    If thousand > 0 Then
        result = result & NumberBelowThousand(thousand) & " Thousand "
    End If

    If hundred > 0 Then
        result = result & NumberBelowThousand(hundred)
    End If

    IndianNumberToWords = Trim(result)

End Function


Private Function NumberBelowThousand(ByVal numberValue As Long) As String

    Dim ones As Variant
    Dim tens As Variant
    Dim result As String

    ones = Array("", "One", "Two", "Three", "Four", "Five", "Six", "Seven", "Eight", "Nine", _
                 "Ten", "Eleven", "Twelve", "Thirteen", "Fourteen", "Fifteen", "Sixteen", _
                 "Seventeen", "Eighteen", "Nineteen")

    tens = Array("", "", "Twenty", "Thirty", "Forty", "Fifty", "Sixty", "Seventy", "Eighty", "Ninety")

    If numberValue >= 100 Then
        result = ones(numberValue \ 100) & " Hundred "
        numberValue = numberValue Mod 100
    End If

    If numberValue >= 20 Then
        result = result & tens(numberValue \ 10) & " "
        numberValue = numberValue Mod 10
    End If

    If numberValue > 0 Then
        result = result & ones(numberValue)
    End If

    NumberBelowThousand = Trim(result)

End Function

Private Sub AddPDFOuter3DBorder(ByVal ws As Worksheet, ByVal lastPrintRow As Long)

    Dim borderRange As Range

    Set borderRange = ws.Range("A1:H" & lastPrintRow)

    'Remove old shape borders
    On Error Resume Next
    ws.Shapes("PDF_Outer_Shadow_1").Delete
    ws.Shapes("PDF_Outer_Shadow_2").Delete
    ws.Shapes("PDF_Outer_Main_Border").Delete
    ws.Shapes("PDF_Outer_Inner_Border").Delete
    On Error GoTo 0

    'Clear only outside border first
    With borderRange
        .Borders(xlEdgeLeft).LineStyle = xlNone
        .Borders(xlEdgeTop).LineStyle = xlNone
        .Borders(xlEdgeRight).LineStyle = xlNone
        .Borders(xlEdgeBottom).LineStyle = xlNone
    End With

    'Apply full outer border
    With borderRange.Borders(xlEdgeLeft)
        .LineStyle = xlContinuous

Private Function CleanFileName(ByVal fileName As String) As String

    fileName = Replace(fileName, "\", "_")

    fileName = Replace(fileName, ":", "_")
    fileName = Replace(fileName, "*", "_")
    fileName = Replace(fileName, "?", "_")
    fileName = Replace(fileName, """", "_")
    fileName = Replace(fileName, "<", "_")
    fileName = Replace(fileName, ">", "_")
    fileName = Replace(fileName, "|", "_")

    If fileName = "" Then fileName = "Invoice"

Private Function BuildInvoiceDetailsQRText(ByVal payAmount As Double) As String

    Dim i As Long
    Dim totalInvoice As Double
    Dim paidAmount As Double
    Dim qrText As String
    Dim paymentStatus As String
    Dim productName As String

    paymentStatus = LCase$(Trim$(Me.Controls("cmbStatus").Value))

    With Me.Controls("lstItems")
        For i = 0 To .ListCount - 1
            totalInvoice = totalInvoice + Val(.List(i, 7))
        Next i
    End With

    paidAmount = Val(Me.Controls("txtPaidAmount").Text)

    If paymentStatus = "paid" Then
        paidAmount = totalInvoice
        payAmount = 0
    ElseIf paymentStatus = "pending" Then
        paidAmount = 0
        payAmount = totalInvoice
    End If

    If paidAmount > totalInvoice Then paidAmount = totalInvoice
    If payAmount < 0 Then payAmount = 0

    ' Short QR text = clear QR image like your sample
    qrText = _
        "INV:" & Me.Controls("txtInvoiceNo").Text & vbLf & _
        "NAME:" & Me.Controls("txtCustomerName").Text & vbLf & _
        "PH:" & Replace(FormatPhoneNumber(Me.Controls("txtPhone").Text), " ", "") & vbLf

    With Me.Controls("lstItems")
        For i = 0 To .ListCount - 1

            productName = .List(i, 1)
            productName = Replace(productName, ",", " ")
            productName = Replace(productName, vbCrLf, " ")
            productName = Replace(productName, vbLf, " ")
            productName = Left$(productName, 18)

            qrText = qrText & _
                "P" & (i + 1) & ":" & productName & _
                "|Q:" & .List(i, 2) & _
                "|R:" & Format(Val(.List(i, 3)), "0") & _
                "|G:" & Format(Val(.List(i, 5)), "0") & _
                "|T:" & Format(Val(.List(i, 7)), "0") & vbLf
        Next i
    End With

    qrText = qrText & _
        "TOTAL:" & Format(totalInvoice, "0") & vbLf & _
        "PAID:" & Format(paidAmount, "0") & vbLf & _
        "BAL:" & Format(payAmount, "0") & vbLf & _
        "STATUS:" & Me.Controls("cmbStatus").Value

    BuildInvoiceDetailsQRText = qrText

End Function

Private Function UrlEncodeText(ByVal textValue As String) As String

    Dim i As Long
    Dim ch As String
    Dim code As Long
    Dim result As String

    For i = 1 To Len(textValue)

        ch = Mid$(textValue, i, 1)
        code = AscW(ch)

        Select Case code
            Case 48 To 57, 65 To 90, 97 To 122
                result = result & ch
            Case 32
                result = result & "%20"
            Case 10
                result = result & "%0A"
            Case 13
                result = result & "%0D"
            Case Else
                result = result & "%" & Right$("0" & Hex(code), 2)
        End Select

    Next i

    UrlEncodeText = result

End Function

Private Function DownloadQRCodeImage(ByVal qrText As String, ByVal invoiceNo As String) As String

    Dim qrURL As String
    Dim qrPath As String
    Dim result As Long
    Dim encodedText As String

    ' Better URL encoding
    On Error Resume Next
    encodedText = Application.WorksheetFunction.EncodeURL(qrText)
    If encodedText = "" Then encodedText = UrlEncodeText(qrText)
    On Error GoTo 0

    ' Plain black and white QR code like your sample image
    qrURL = "https://quickchart.io/qr?" & _
            "text=" & encodedText & _
            "&size=1000" & _
            "&margin=1" & _
            "&ecLevel=L" & _
            "&dark=000000" & _
            "&light=ffffff" & _
            "&format=png"

    qrPath = Environ$("TEMP") & "\" & CleanQRFileName(invoiceNo & "_Invoice_QR") & ".png"

    On Error Resume Next
    Kill qrPath
    On Error GoTo 0

    result = URLDownloadToFile(0, qrURL, qrPath, 0, 0)

    If result = 0 And Dir(qrPath) <> "" Then
        DownloadQRCodeImage = qrPath
    Else
        MsgBox "QR image download failed. Please check internet connection.", vbCritical, "QR Error"
        DownloadQRCodeImage = ""
    End If

End Function

Private Function FormatPhoneNumber(ByVal phoneNo As String) As String

    phoneNo = Trim(phoneNo)
    phoneNo = Replace(phoneNo, " ", "")

    If Left(phoneNo, 3) = "+91" Then
        FormatPhoneNumber = phoneNo
    ElseIf Left(phoneNo, 2) = "91" And Len(phoneNo) = 12 Then
        FormatPhoneNumber = "+" & phoneNo
    ElseIf phoneNo = "" Then
        FormatPhoneNumber = ""
    Else
        FormatPhoneNumber = "+91 " & phoneNo
    End If

End Function
